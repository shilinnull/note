题目一：【用队列实现栈】https://leetcode.cn/problems/implement-stack-using-queues/description/     "字节-2024-开发 谷歌-2024-开发"

```CPP
class MyStack {
public:
    queue<int> q;
    MyStack() {
        
    }
    
    void push(int x) {
        int size = q.size();
        q.push(x);
        while(size--) {
            int num = q.front(); q.pop();
            q.push(num);
        }
    }
    
    int pop() {
        int ret = q.front(); q.pop();
        return ret;
    }
    
    int top() {
        return q.front();
    }
    
    bool empty() {
        return q.empty();
    }
};
```



题目二：【数青蛙】https://leetcode.cn/problems/minimum-number-of-frogs-croaking/         "华为-2024-开发 特斯拉-2024-开发" 

```cpp
class Solution {
public:
    int minNumberOfFrogs(string croakOfFrogs) {
        string s = "croak";
        unordered_map<char, int> index;     // 用来保存对应字符的下标

        int n = s.size();
        vector<int> hash(n);    // 模拟
        for(int i = 0; i < n; i++)
            index[s[i]] = i;    // 对应的下标
            
        for(auto ch : croakOfFrogs) {
            if(ch == 'c') {
                if(hash[n - 1] != 0) hash[n - 1]--;
                hash[0]++;
            } else {
                int i = index[ch];
                // 前一个字符如果没有就返回-1
                if(hash[i - 1] == 0) return -1; 
                hash[i - 1]--;  // 找到前一个字符--
                hash[i]++;      // 当前字符++
            }
        }
        for(int i = 0; i < n - 1; i++) {
            if(hash[i]) return -1;
        }
        return hash[n - 1];
    }
};
```

```cpp
class Solution {
public:
    int minNumberOfFrogs(string croakOfFrogs) {
        int hash[5] = {0};
        // 0   1   2   3   4
        // c   r   o   a   k
        int n = croakOfFrogs.size();
        for(int i = 0; i < n; i++) {
            if(croakOfFrogs[i] == 'c') {
                if(hash[4] > 0) {
                    hash[4]--;
                    hash[0]++;
                    continue;
                }
                hash[0]++;
            }
            else if(croakOfFrogs[i] == 'r') {
                if(hash[0] <= 0) return -1;
                hash[0]--;
                hash[1]++;
            } else if(croakOfFrogs[i] == 'o') {
                if(hash[1] <= 0) return -1;
                hash[1]--;
                hash[2]++;
            }else if(croakOfFrogs[i] == 'a') {
                if(hash[2] <= 0) return -1;
                hash[2]--;
                hash[3]++;
            }else if(croakOfFrogs[i] == 'k') {
                if(hash[3] <= 0) return -1;
                hash[3]--;
                hash[4]++;
            }
        }
        for(int i = 0; i < 4; i++) {
            if(hash[i]){
                return -1;
            }
        }
        return hash[4];
    }
};
```
