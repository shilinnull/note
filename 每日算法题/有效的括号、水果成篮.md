题目一：【有效的括号】https://leetcode.cn/problems/valid-parentheses/description/     "网易-2024-开发  蔚来-2023-开发  京东-2023-开发"

```c++
class Solution {
public:
    bool isValid(string s) {
        stack<char> st;
        unordered_map<char, char> hash{{')', '('}, {']', '['}, {'}', '{'}};
        int i = 0;
        while(i < s.size()) {
            if(s[i] == '(' || s[i] == '[' || s[i] == '{') st.push(s[i]);
            else 
            {
                if(st.empty()) return false;
                char top = st.top(); st.pop();
                if(top != hash[s[i]]) return false;
            }
            i++;
        }
        return st.empty();
    }
};
```



题目二：【水果成篮】https://leetcode.cn/problems/fruit-into-baskets/description/         "Meta-2024-开发  微软-2024-开发"

```c++
class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<int, int> hash;
        int left = 0, right = 0, len = 0;
        while(right < fruits.size()) {
            hash[fruits[right]]++;  // 进窗口
            while(hash.size() > 2) {    // 判断
                hash[fruits[left]]--; // 出窗口
                if(hash[fruits[left]] == 0) hash.erase(fruits[left]);
                left++;
            }
            // 更新结果
            len = max(len, right - left + 1);
            right++;
        }
        return len;
    }
};
```
