题目一：【表达式求值】https://www.nowcoder.com/practice/c215ba61c8b1443b996351df929dc4d4?tpId=196&tqId=37183&rp=1&ru=/exam/oj&qru=/exam/oj&sourceUrl=%2Fexam%2Foj%3FquestionJobId%3D10%26subTabName%3Donline_coding_page&difficulty=undefined&judgeStatus=undefined&tags=581&title= "网易-2024-开发 深信服-2023-测开"  

**思路**：

- 使用两个栈：`nums` 存储数字，`ops` 存储运算符。
- 遍历字符串，按以下规则处理：
  1. 遇到数字：解析完整的数字并入栈。
  2. 遇到 `(`：直接入运算符栈。
  3. 遇到 `)`：计算括号内的表达式，直到遇到 `(`。
  4. 遇到运算符 `+ - *`：在入栈前，如果栈顶运算符优先级 >= 当前运算符，则先计算栈顶的运算，再入栈。
- 最后，将栈中剩余运算符依次计算，数字栈的栈顶即为结果。

**优先级定义**：

- `+`、`-` 优先级为 1
- `*` 优先级为 2
- `(` 优先级为 0（直接入栈，不触发计算）

```CPP
class Solution {
public:
    int solve(string s) {
        stack<int> nums;
        stack<char> ops;
        unordered_map<char, int> priority = {
            {'+', 1},
            {'-', 1},
            {'*', 2},
            {'(', 0}
        };
        
        int n = s.size();
        for (int i = 0; i < n; i++) {
            char c = s[i];
            if (c == ' ') continue;
            
            if (isdigit(c)) {
                int num = 0;
                while (i < n && isdigit(s[i])) {
                    num = num * 10 + (s[i] - '0');
                    i++;
                }
                nums.push(num);
                i--; // 回退一步，因为外层循环会再 i++
            } else if (c == '(') {
                ops.push(c);
            } else if (c == ')') {
                // 计算括号内的表达式
                while (!ops.empty() && ops.top() != '(') {
                    calc(nums, ops);
                }
                ops.pop(); // 弹出 '('
            } else { // 运算符 + - *
                // 当前运算符优先级 <= 栈顶优先级时，先计算栈顶
                while (!ops.empty() && priority[ops.top()] >= priority[c]) {
                    calc(nums, ops);
                }
                ops.push(c);
            }
        }
        
        // 处理剩余的运算符
        while (!ops.empty()) {
            calc(nums, ops);
        }
        
        return nums.top();
    }
    
private:
    void calc(stack<int>& nums, stack<char>& ops) {
        if (nums.size() < 2 || ops.empty()) return;
        int b = nums.top(); nums.pop();
        int a = nums.top(); nums.pop();
        char op = ops.top(); ops.pop();
        int res = 0;
        if (op == '+') res = a + b;
        else if (op == '-') res = a - b;
        else if (op == '*') res = a * b;
        nums.push(res);
    }
};
```



题目二：【非递减子序列】https://leetcode.cn/problems/non-decreasing-subsequences/description/ "字节-2024-开发 雅虎-2024-开发"

- 使用 DFS 递归枚举所有可能的子序列。
- 用 `set`（或数组哈希）在**同一层**去重，避免相同数字重复选取。
- 保证子序列是**非递减**的：只有当前数字 >= 路径最后一个数字时才能加入。

```CPP
class Solution {
    vector<vector<int>> ret;
    vector<int> path;
public:
    vector<vector<int>> findSubsequences(vector<int>& nums) {
        dfs(nums, 0);
        return ret;
    }
    void dfs(vector<int>& nums, int pos) {
        if(path.size() >= 2) {
            ret.push_back(path);
        }
        unordered_set<int> used; // 用于当前层去重
        for(int i = pos; i < nums.size(); i++) {
            // 如果当前数字在本层已经用过，跳过
            if (used.count(nums[i])) continue;
            // 保证非递减：如果path为空 或者 nums[i] >= 最后一个元素
            if(path.empty() || nums[i] >= path.back()) {
                used.insert(nums[i]);
                path.push_back(nums[i]);
                dfs(nums, i + 1);
                path.pop_back();
            }
        }
    }
};
```
