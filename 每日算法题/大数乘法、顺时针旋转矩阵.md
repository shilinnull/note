class Solution {
public:
    string solve(string s, string t) {
        reverse(s.begin(), s.end());
        reverse(t.begin(), t.end());
        int m = s.size(), n = t.size();
        vector<int> sum(m + n);
        for(int i = 0; i < m; i++) {
            for(int j = 0; j < n; j++) {
                sum[i + j] += (s[i] - '0') * (t[j] - '0');
            }
        }
        // 处理进位
        string ret;
        int next = 0, i = 0;
        while(i < sum.size() || next) {
            if(i < sum.size()) next += sum[i];
            ret += next % 10 + '0';
            next /= 10;
            i++;
        }
        while(ret.size() > 1 && ret.back() == '0') ret.pop_back();
        reverse(ret.begin(), ret.end());
        return ret;
    }
};

#include <vector>
class Solution {
public:
    vector<vector<int> > rotateMatrix(vector<vector<int> >& mat, int n) {
        vector<vector<int> > ret(n, vector<int>(n));
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < n; j++) {
                ret[j][n - i - 1] = mat[i][j];
            }
        }
        return ret;
    }
};
