题目一：【全排列Ⅱ】https://leetcode.cn/problems/permutations-ii/?envType=problem-list-v2&envId=array          米哈游-2024-测开        

```cpp
class Solution {
    vector<int> path;
    vector<vector<int>> ret;
    bool check[9]{};
public:
    vector<vector<int>> permuteUnique(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        int pos = 0;
        dfs(nums, pos);
        return ret;
    }
    void dfs(vector<int>& nums, int pos)
    {
        if(path.size() == nums.size()) 
        {
            ret.push_back(path);
            return;
        }

        for(int i = 0; i < nums.size(); i++) {
            // 1. 当前位置被使用了
            // 2. 检查i-1的时候i==0时会越界 && 检查i - 1位置是否与当前位置重复 && i-1位置需为没被访问过
            if(check[i] || (i != 0 && nums[i] == nums[i - 1] && !check[i - 1]))
                continue;
            check[i] = true;
            path.push_back(nums[i]);
            dfs(nums, pos + 1);
            path.pop_back();
            check[i] = false;
        }
    }
};
```



题目二：【盛最多水的容器】https://leetcode.cn/problems/container-with-most-water/description/                       饿了么-2024-开发

```CPP
class Solution {
public:
    int maxArea(vector<int>& height) {
        int maxVal = 0;
        for(int left = 0, right = height.size() - 1; left < right;) {
            maxVal = max(maxVal, (right - left) * min(height[left], height[right]));
            if(height[left] < height[right]) left++;
            else right--;
        }
        return maxVal;
    }
};
```
