题目一：【最短无序连续子数组】 https://leetcode.cn/problems/shortest-unsorted-continuous-subarray/description/   "字节-2024-开发  谷歌-2024-开发"

```CPP
class Solution {
public:
    int findUnsortedSubarray(vector<int>& nums) {
        int n = nums.size();  
        if (n <= 1) return 0;
        vector<int> sorted_nums = nums;
        sort(sorted_nums.begin(), sorted_nums.end());
        
        int left = 0, right = n - 1;
        while (left < n && nums[left] == sorted_nums[left]) left++;
        while (right >= 0 && nums[right] == sorted_nums[right]) right--;
        
        return max(0, right - left + 1);
    }
};
```

```CPP
class Solution {
public:
    int findUnsortedSubarray(vector<int>& nums) {
        int n = nums.size();
        if (n <= 1) return 0;
        int min_val = INT_MAX, max_val = INT_MIN;
        // 从左到右找到无序部分的最小值
        for (int i = 1; i < n; i++) {
            if (nums[i] < nums[i - 1]) {
                min_val = min(min_val, nums[i]);
            }
        }
        
        // 从右到左找到无序部分的最大值
        for (int i = n - 2; i >= 0; i--) {
            if (nums[i] > nums[i + 1]) {
                max_val = max(max_val, nums[i]);
            }
        }
        
        // 如果数组已经有序
        if (min_val == INT_MAX && max_val == INT_MIN) return 0;
        
        // 找到最小值应该插入的位置
        int left = 0;
        while (left < n && nums[left] <= min_val) left++;
        
        // 找到最大值应该插入的位置
        int right = n - 1;
        while (right >= 0 && nums[right] >= max_val) right--;
        
        return right - left + 1;
    }
};
```





题目二：【接雨水】 https://leetcode.cn/problems/trapping-rain-water/description/   "字节-2024-开发  腾讯-2024-开发" 

```CPP
class Solution {
public:
    int trap(vector<int>& height) {
        int left = 0, right = height.size() - 1;
        int lmax = 0, rmax = 0;
        int ret = 0;
        // 每个格子能承载的雨水量为该格子左边的最大值lmax和右边的最大值rmax二者中最小的
        while(left <= right){
            lmax = max(lmax, height[left]);
            rmax = max(rmax, height[right]);
            if(height[left] < height[right]) {
                ret += lmax - height[left];
                left++;
            }else {
                ret += rmax - height[right];
                right--;
            }
        }
        return ret;
    }
};
```
