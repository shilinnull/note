题目一：【二叉树展开为链表】https://leetcode.cn/problems/flatten-binary-tree-to-linked-list/description/         "亚马逊-2024-开发 微软-2024-开发"

```CPP
class Solution {
    TreeNode *p = nullptr;
public:
    void flatten(TreeNode* root) {
        if(!root) return ;
        TreeNode* left = root->left;
        TreeNode* right = root->right;
        if(p) {
            p->left = nullptr;
            p->right = root;
        }
        p = root;
        flatten(left);
        flatten(right);
    }
};
```

题目二：【提莫攻击】https://leetcode.cn/problems/teemo-attacking/description/        "字节-2024-开发 Riot games-2024-开发”

```CPP
class Solution {
public:
    int findPoisonedDuration(vector<int>& timeSeries, int duration) {
        int ret = 0;
        for(int i = 1; i < timeSeries.size(); i++) {
            if(timeSeries[i] - timeSeries[i - 1] >= duration) ret += duration; // 上次中毒可以持续duration
            else ret += timeSeries[i] - timeSeries[i - 1];   // 上次的中毒只能持续两者的差值
        }
        return ret + duration;  // 加上最后一个
    }
};
```
