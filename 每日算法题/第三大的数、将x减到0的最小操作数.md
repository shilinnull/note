题目一：【第三大的数】        https://leetcode.cn/problems/third-maximum-number/description/         亚马逊-2024-开发

```cpp
class Solution {
public:
    int thirdMax(vector<int>& nums) {
        set<int> s;
        for(int n : nums) {
            s.insert(n);
            if(s.size() > 3) s.erase(s.begin());
        }
        // 如果剩余元素小于3了就返回最大值
        return s.size() == 3 ? *s.begin() : *(--s.end());  
    }
};
```



题目二：【将x减到0的最小操作数】https://leetcode.cn/problems/minimum-operations-to-reduce-x-to-zero/description/        "谷歌-2024-开发   字节跳动-2024-开发"

正难则反

找中间的一段连续然后让它等于`sum - x`

```CPP
class Solution {
public:
    int minOperations(vector<int>& nums, int x) {
        int sum = 0, len = -1;
        for(int n : nums) sum += n;
        int target = sum - x;
        if(target < 0) return -1;	// 如果小于0了就不需要进行处理
        for(int left = 0, right = 0, tmp = 0; right < nums.size(); right++){
            // 进窗口
            tmp += nums[right];
            while(tmp > target) tmp -= nums[left++];    // 判断+出窗口
            if(tmp == target) len = max(len, right - left + 1); // 需要恰好等于target
        }

        return len == -1 ? len : nums.size() - len;
    }
};
```
