题目一：【用栈实现队列】        https://leetcode.cn/problems/implement-queue-using-stacks/description/        "字节-2024-开发 微软-2024-开发"

```c++
class MyQueue {
    stack<int> st1;
    stack<int> st2;
public:
    MyQueue() {}
    
    void push(int x) {
        st1.push(x);
    }
    
    int pop() {
        int top = peek();
        st2.pop();
        return top;
    }
    
    int peek() {
        if(!st2.empty()) return st2.top();
        if(st1.empty()) return -1;
        while(!st1.empty()) { st2.push(st1.top()); st1.pop();}
        return st2.top();
    }
    
    bool empty() {
        return st1.empty() && st2.empty();
    }
};
```



题目二：【递归乘法】        https://leetcode.cn/problems/recursive-mulitply-lcci/description/         去哪儿网-2022-开发 

```c++
class Solution {
public:
    int multiply(int A, int B) {
        if(B == 1) return A;
        return A + multiply(A, B - 1);
    }
};
```
