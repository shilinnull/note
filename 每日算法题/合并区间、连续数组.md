```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        sort(intervals.begin(), intervals.end());
        int a = intervals[0][0], b = intervals[0][1];
        vector<vector<int>> ret;
        for(int i = 0; i < intervals.size(); i++) {
            if(b < intervals[i][0]) {
                ret.push_back({a, b});
                a = intervals[i][0];
                b = intervals[i][1];
            }
            else b = max(b, intervals[i][1]);
        }
        ret.push_back({a, b});
        return ret;
    }
};
```
