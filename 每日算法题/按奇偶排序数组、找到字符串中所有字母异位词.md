题目一：【按奇偶排序数组】https://leetcode.cn/problems/sort-array-by-parity/description/      猿辅导-2021-开发

```cpp
class Solution {
public:
    vector<int> sortArrayByParity(vector<int>& nums) {
        for(int left = 0, right = nums.size() - 1; left < right;) {
            while(left < right && nums[left] % 2 == 0) left++;
            while(left < right && nums[right] % 2 == 1) right--;
            swap(nums[left++], nums[right--]);
        }
        return nums;
    }
};
```



题目二：【找到字符串中所有字母异位词】https://leetcode.cn/problems/VabMRr/description/   图森-2023-测开

```c++
class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        vector<int> ret;
        int hash1[128] {}, hash2[128] {};
        for(auto ch : p) hash1[ch]++;

        for(int left = 0, right = 0, count = 0; right < s.size(); right++) {
            char in = s[right];
            hash2[in]++;
            if(hash2[in] <= hash1[in]) count++;     // 有效字符
            if(right - left + 1 > p.size()) {
                char out = s[left++];
                if(hash2[out] <= hash1[out]) count--; // 有效字符
                hash2[out]--;
            }
            if(count == p.size()) 
                ret.push_back(left);        // 更新结果
        }
        return ret;
    }
};
```
