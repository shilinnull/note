题目一：【快乐数】https://leetcode.cn/problems/happy-number/ 

快慢指针实现

```CPP
class Solution {
public:
    int getNext(int n) {
        int sum = 0;
        while (n) {
            int d;
            d = n % 10;
            n /= 10;
            sum += d * d;
        }
        return sum;
    }
    bool isHappy(int n) {
        int slow = n, fast = getNext(n);

        // 想要退出循环，要么快慢相遇，代表成环；要么快指针为1，代表为true
        while (slow != fast && fast != 1) {
            slow = getNext(slow);
            fast = getNext(getNext(fast));
        }
        return (fast == 1);
    }
};
```

class Solution {
public:
    bool isHappy(int n) {
        for (int i = 0; i < 100; i++) {
            int ans = 0;
            while (n > 0) {
                ans += (n % 10) * (n % 10);
                n = n / 10;
            }
            n = ans;
            if (n == 1)
                return true;
        }
        return false;
    }
};
