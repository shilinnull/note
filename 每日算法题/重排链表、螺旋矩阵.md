题目一：【重排链表】https://leetcode.cn/problems/reorder-list/description/   "拼多多-2024-开发 百度-2024-开发 美团-2024-开发"

```CPP
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        if(!head || !head->next) return head;
        ListNode* p1 = nullptr, *p2 = head, *p3 = head->next;
        while(p2) {
            p2->next = p1;
            p1 = p2;
            p2 = p3;
            if(p3) p3 = p3->next;
        }
        return p1;
    }

    void reorderList(ListNode* head) {
        // 1. 找中间节点
        ListNode* slow = head, * fast = head;
        while(fast->next && fast->next->next != nullptr) {
            slow = slow->next;
            fast = fast->next->next;
        }
        // 2. 反转后半段
        ListNode* t = slow->next;
        slow->next = nullptr;

        ListNode* cur1 = head;
        ListNode* cur2 = reverseList(t);
        // 3. 插入到head每个节点的空隙
        while(cur1 && cur2) {
            // 记录cur2的下一个节点
            ListNode* cur2next = cur2;
            cur2 = cur2->next;

            // 插入节点
            ListNode* cur1next = cur1->next;
            cur2next->next = cur1->next;
            cur1->next = cur2next;
            
            // 更新
            cur1 = cur1next;
        }
    }
};
```

题目二：【螺旋矩阵】https://leetcode.cn/problems/spiral-matrix/description/  

```CPP
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        vector<int> ret;
        if (matrix.empty() || matrix[0].empty()) return ret;
        
        int top = 0, bottom = matrix.size() - 1;
        int left = 0, right =  matrix[0].size() - 1;
        while (top <= bottom && left <= right) {
            // 从左到右
            for (int i = left; i <= right; i++)
                ret.push_back(matrix[top][i]);
            top++;

            // 从上到下
            for(int i = top; i <= bottom; i++) 
                ret.push_back(matrix[i][right]);
            right--;

            // 检测遍历是否完成
            if(left > right || top > bottom) break;

            // 从右到左
            for(int i = right; i >= left; i--)
                ret.push_back(matrix[bottom][i]);
            bottom--;
            
            // 从下到上
            for(int i = bottom; i >= top; i--) 
                ret.push_back(matrix[i][left]);
            left++;
        }
        return ret;
    }
};
```
