题目一：【反转字符串】 https://leetcode.cn/problems/reverse-string/description/         "Meta-2024-开发  字节跳动-2023-开发"

```cpp
class Solution {
public:
    void reverseString(vector<char>& s) {
        for(int i = 0, j = s.size() - 1; i < s.size() / 2; i++, j--) {
            swap(s[i], s[j]);
        }
    }
};
```



题目二：【最小覆盖子串】：https://leetcode.cn/problems/M1oyTv/description/             "字节跳动-2024-开发"

```cpp
class Solution {
public:
    string minWindow(string s, string t) {
        int hash1[128] {}, hash2[128]{};
        int count = 0;
        for(auto ch : t) if(hash1[ch]++ == 0) count++;

        int left = 0, right = 0, cnt = 0, begin = 0, len = INT_MAX;
        while(right < s.size()) {
            // 进窗口
            char in = s[right];
            if(++hash2[in] == hash1[in]) cnt++;
            while(cnt == count) { // 判断
                // 更新结果
                if(right - left + 1 < len) {
                    begin = left;
                    len = right - left + 1;
                }

                // 出窗口
                char out = s[left++];
                if(hash2[out]-- == hash1[out]) cnt--;
            }
            right++;
        }
        return len == INT_MAX ? "" : s.substr(begin, len);
    }
};
```
