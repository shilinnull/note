题目一：【反转字符串】 https://leetcode.cn/problems/reverse-string/description/         "Meta-2024-开发  字节跳动-2023-开发"

```cpp
class Solution {
public:
    void reverseString(vector<char>& s) {
        for(int i = 0, j = s.size() - 1; i < s.size() / 2; i++, j--) {
            swap(s[i], s[j]);
        }
    }
};
```
