题目一：【从中序后续遍历构造二叉树】  https://leetcode.cn/problems/construct-binary-tree-from-inorder-and-postorder-traversal/?envType=problem-list-v2&envId=tree         诺瓦-2023-后端开发

```cpp
class Solution {
    private int prei;

    public TreeNode buildTree(int[] inorder, int[] postorder) {
        prei = postorder.length - 1; // 从最后面开始
        return _buildTree(inorder, postorder, 0, postorder.length - 1);
    }

    public TreeNode _buildTree(int[] inorder, int[] postorder, int inbegin, int inend) {
        if (inbegin > inend)
            return null;

        TreeNode root = new TreeNode(postorder[prei--]); // 后序确定根
        
        // 分割左右区间
        int rooti = inbegin;
        while (rooti <= inend) {
            if (inorder[rooti] == root.val) {
                break;
            } else {
                rooti++;
            }
        }

        // 先构建右树，再构建左树
        root.right = _buildTree(inorder, postorder, rooti + 1, inend);
        root.left = _buildTree(inorder, postorder, inbegin, rooti - 1);

        return root;
    }
}
```
